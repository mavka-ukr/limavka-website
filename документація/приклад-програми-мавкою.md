# Приклад програми Мавкою

```мавка
кольори = [
  Колір.БІЛИЙ,
  Колір.ЧЕРВОНИЙ,
  Колір.ЗЕЛЕНИЙ,
  Колір.СИНІЙ,
]

ікс = Полотно.Ш/2
ігр = Полотно.В/2
р = 20
пк = 0
колір = кольори[пк]

запущено = дійсне

поки запущено
  Полотно.заповнити(0)
  Полотно.круг(ікс, ігр, р, колір)
  Полотно.запланувати() ;; запланувати відмалювання

  Кнопки.оновити() ;; отримати стан кнопок

  якщо Кнопки.вверх_нажато
    ігр -= 1
  кінець
  якщо Кнопки.вниз_нажато
    ігр += 1
  кінець
  якщо Кнопки.вліво_нажато
    ікс -= 1
  кінець
  якщо Кнопки.вправо_нажато
    ікс += 1
  кінець

  якщо Кнопки.а_нажато
    р += 1
  кінець
  якщо Кнопки.д_нажато
    р -= 1
  кінець

  якщо Кнопки.г_щойно_нажато
    пк += 1
    якщо пк == кольори.довжина
      пк = 0
    кінець

    колір = кольори[пк]
  кінець

  якщо Кнопки.б_щойно_відпущено
    запущено = недійсне
  кінець
кінець
```
