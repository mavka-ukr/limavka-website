# Приклад програми Мавкою

```мавка
кольори = [
  Колір.БІЛИЙ,
  Колір.ЧЕРВОНИЙ,
  Колір.ЗЕЛЕНИЙ,
  Колір.СИНІЙ,
]

ікс = Полотно.Ш/2
ігр = Полотно.В/2
р = 20
пк = 0
колір = кольори[пк]
шв = 50

запущено = дійсне
ост_ч = отримати_час()

кнс_ч = отримати_час()
кіл_кдр = 0
пот_кнс = 0

поки запущено
  пот_ч = отримати_час()

  різ_ч = пот_ч - ост_ч
  ост_ч = пот_ч


  оновити_кнопки() ;; оновити стан кнопок

  якщо Кнопки.вверх.нажато
    ігр -= (шв * (різ_ч / 1000))
  кінець
  якщо Кнопки.вниз.нажато
    ігр += (шв * (різ_ч / 1000))
  кінець
  якщо Кнопки.вліво.нажато
    ікс -= (шв * (різ_ч / 1000))
  кінець
  якщо Кнопки.вправо.нажато
    ікс += (шв * (різ_ч / 1000))
  кінець

  якщо Кнопки.а.нажато
    р += (шв * (різ_ч / 1000))
  кінець
  якщо Кнопки.д.нажато
    р -= (шв * (різ_ч / 1000))
  кінець

  якщо Кнопки.г.щойно_нажато
    пк += 1
    якщо пк == кольори.довжина
      пк = 0
    кінець

    колір = кольори[пк]
  кінець

  якщо Кнопки.б.щойно_відпущено
    запущено = недійсне
  кінець


  Полотно.заповнити(0)
  Полотно.круг(ікс, ігр, р, колір)

  кіл_кдр += 1
  якщо (пот_ч - кнс_ч) >= 1000
    пот_кнс = кіл_кдр
    кіл_кдр = 0
    кнс_ч = пот_ч
  кінець

  Полотно.текст(30, 30, Текст(пот_кнс), розмір=3, колір=Колір.ЧЕРВОНИЙ)

  Полотно.відмалювати() ;; запланувати відмалювання

  ч_обр = отримати_час() - пот_ч
  трв_кдр = 1000.0 / Полотно.КНС

  якщо ч_обр < трв_кдр
    спати(трв_кдр - ч_обр)
  кінець
кінець
```
